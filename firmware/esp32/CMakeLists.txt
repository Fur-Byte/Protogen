cmake_minimum_required(VERSION 3.16)
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
if(SDKCONFIG MATCHES "paw$")
  list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults.paw")
endif()
if(DEFINED PROTOGEN_NATIVE_ROLE)
  if(NOT PROTOGEN_NATIVE_ROLE MATCHES "^(helmet|paw)$")
    message(FATAL_ERROR "Invalid native Protogen role")
  endif()
  set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/src")
  set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
  if(PROTOGEN_NATIVE_ROLE STREQUAL "paw")
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults.paw")
  endif()
  if(DEFINED PROTOGEN_NATIVE_EXTRA_DEFAULTS)
    list(APPEND SDKCONFIG_DEFAULTS "${PROTOGEN_NATIVE_EXTRA_DEFAULTS}")
  endif()
endif()

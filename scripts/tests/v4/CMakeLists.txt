cmake_minimum_required(VERSION 3.20)
project(ProtogenExperimentalV4 C CXX)
option(SANITIZE "Instrument host code and crypto dependencies" ON)
set(HOST_SANITIZERS "address,undefined" CACHE STRING "Required host sanitizer selection")
if(SANITIZE)
  add_compile_options(-fsanitize=${HOST_SANITIZERS} -fno-sanitize-recover=all)
  add_link_options(-fsanitize=${HOST_SANITIZERS})
endif()
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
set(TF_PSA_CRYPTO_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../wireless-crypto-config.h" CACHE FILEPATH "" FORCE)
set(MBEDTLS_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../wireless-config.h" CACHE FILEPATH "" FORCE)
if(NOT MBEDTLS_SOURCE)
  message(FATAL_ERROR "Set MBEDTLS_SOURCE to pinned ESP-IDF Mbed TLS source")
endif()
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/../wireless/compat")
add_subdirectory(${MBEDTLS_SOURCE} mbedtls)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
set(V4 "${ROOT}/firmware/esp32/protocol_v4")
add_library(v4 STATIC "${V4}/Core.cpp" "${V4}/Derivation.cpp" "${V4}/Fragments.cpp" "${V4}/Storage.cpp" "${V4}/Traffic.cpp" "${V4}/Worker.cpp" "${V4}/HostCrypto.cpp" "${V4}/Native.c" "${ROOT}/firmware/esp32/src/WirelessCrypto.cpp")
target_compile_features(v4 PRIVATE cxx_std_17 c_std_11)
target_include_directories(v4 PUBLIC "${V4}" "${ROOT}/firmware/PROT_OS7_MAX7219/src/HelmetProtocol" PRIVATE "${ROOT}/firmware/esp32/src")
set_source_files_properties("${V4}/Native.c" PROPERTIES COMPILE_DEFINITIONS "MLK_CONFIG_FILE=\"${V4}/NativeConfig.h\"")
target_link_libraries(v4 PUBLIC mbedtls)
target_compile_options(v4 PRIVATE -Wall -Wextra -Werror -fstack-usage)
option(V4_CORE_ASSERTIONS "Debug assertions in core (tests always assert)" ON)
if(V4_CORE_ASSERTIONS)
  target_compile_options(v4 PRIVATE -UNDEBUG)
else()
  target_compile_options(v4 PRIVATE -DNDEBUG)
endif()
add_executable(test-v4 test.cpp Checks.cpp Vectors.cpp Qualification.cpp UartIntegration.cpp)
target_include_directories(test-v4 PRIVATE "${ROOT}/firmware/esp32/platform_research")
target_link_libraries(test-v4 PRIVATE v4)
target_compile_options(test-v4 PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_compile_definitions(test-v4 PRIVATE V4_VECTOR_FILE="${CMAKE_CURRENT_SOURCE_DIR}/vectors.json")

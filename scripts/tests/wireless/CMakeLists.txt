cmake_minimum_required(VERSION 3.20)
project(ProtogenWirelessTest C CXX)
option(SANITIZE "Instrument host code and crypto dependencies" ON)
set(HOST_SANITIZERS "address,undefined" CACHE STRING "Required host sanitizer selection")
if(SANITIZE)
  add_compile_options(-fsanitize=${HOST_SANITIZERS} -fno-sanitize-recover=all)
  add_link_options(-fsanitize=${HOST_SANITIZERS})
endif()
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
set(TF_PSA_CRYPTO_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../wireless-crypto-config.h" CACHE FILEPATH "" FORCE)
# Use a crypto-only Mbed TLS configuration; DTLS is tested independently.
set(MBEDTLS_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../wireless-config.h" CACHE FILEPATH "" FORCE)
if(NOT MBEDTLS_SOURCE)
  message(FATAL_ERROR "Set MBEDTLS_SOURCE to the ESP-IDF bundled Mbed TLS 4 source")
endif()
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/compat")
add_subdirectory(${MBEDTLS_SOURCE} mbedtls)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
set(SRC "${ROOT}/firmware/esp32/src")
add_executable(test-wireless ../wireless.cpp "${SRC}/WirelessCrypto.cpp" "${SRC}/WirelessSession.cpp"
  "${SRC}/WirelessPairing.cpp" "${SRC}/SessionHandshake.cpp" "${SRC}/IdentityRecovery.cpp"
  "${SRC}/WirelessLink.cpp" "${SRC}/PairingIndicator.cpp")
target_include_directories(test-wireless PRIVATE "${SRC}" "${ROOT}/firmware/PROT_OS7_MAX7219/src/HelmetProtocol")
target_link_libraries(test-wireless PRIVATE mbedtls)
target_compile_features(test-wireless PRIVATE cxx_std_17)
target_compile_options(test-wireless PRIVATE -Wall -Wextra -Werror -UNDEBUG)

add_executable(test-storage ../storage-faults.cpp)
target_include_directories(test-storage PRIVATE "${SRC}" "${ROOT}/firmware/PROT_OS7_MAX7219/src/HelmetProtocol")
target_compile_features(test-storage PRIVATE cxx_std_17)
target_compile_options(test-storage PRIVATE -Wall -Wextra -Werror -UNDEBUG -DHELMET_RECEIVER)

set(TEENSY "${ROOT}/firmware/PROT_OS7_MAX7219")
add_library(uart-mbed STATIC ../uart-mbed-adapter.cpp
 "${TEENSY}/src/UartCrypto/md.c" "${TEENSY}/src/UartCrypto/sha256.c" "${TEENSY}/src/UartCrypto/platform_util.c")
target_include_directories(uart-mbed PRIVATE "${TEENSY}" "${TEENSY}/src/UartCrypto" "${CMAKE_CURRENT_SOURCE_DIR}/../uart-fixture")
target_compile_definitions(uart-mbed PRIVATE PROTOGEN_UART_AUTH=1)
target_compile_options(uart-mbed PRIVATE -Wall -Wextra -Werror)
add_executable(test-uart ../uart-auth.cpp "${SRC}/WirelessCrypto.cpp")
target_include_directories(test-uart PRIVATE "${SRC}" "${TEENSY}/src/HelmetProtocol")
target_compile_definitions(test-uart PRIVATE PROTOGEN_UART_AUTH=1)
target_compile_features(test-uart PRIVATE cxx_std_17)
target_compile_options(test-uart PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(test-uart PRIVATE uart-mbed mbedtls)

add_executable(benchmark-wireless ../wireless-benchmark.cpp "${SRC}/WirelessCrypto.cpp" "${SRC}/WirelessSession.cpp" "${SRC}/SessionHandshake.cpp")
target_include_directories(benchmark-wireless PRIVATE "${SRC}" "${TEENSY}/src/HelmetProtocol")
target_compile_features(benchmark-wireless PRIVATE cxx_std_17)
target_compile_options(benchmark-wireless PRIVATE -Wall -Wextra -Werror -UNDEBUG -fstack-usage)
target_link_libraries(benchmark-wireless PRIVATE mbedtls)

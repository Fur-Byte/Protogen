cmake_minimum_required(VERSION 3.20)
project(ProtogenDtlsTest C CXX)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
set(MBEDTLS_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../dtls-config.h" CACHE FILEPATH "" FORCE)
set(TF_PSA_CRYPTO_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../dtls-crypto-config.h" CACHE FILEPATH "" FORCE)
if(NOT MBEDTLS_SOURCE)
  message(FATAL_ERROR "Set MBEDTLS_SOURCE to the ESP-IDF bundled Mbed TLS 4 source")
endif()
add_subdirectory(${MBEDTLS_SOURCE} mbedtls)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
add_executable(test-dtls ../dtls.cpp "${ROOT}/firmware/esp32/src/SecureSession.cpp")
target_include_directories(test-dtls PRIVATE "${ROOT}/firmware/esp32/src")
target_link_libraries(test-dtls PRIVATE mbedtls)
target_compile_features(test-dtls PRIVATE cxx_std_17)
target_compile_options(test-dtls PRIVATE -Wall -Wextra -Werror -UNDEBUG)
